P02

P02 · Enquadramento em pipe real; extensão TCP opcional

Como bytes viram mensagens?

Base conceitual: Material da unidade 2 (PDF, nova aba). Tempo de referência: 60–90 min, ajustável pelo professor.

O agregador deve reconhecer um registro mesmo quando o fluxo chega em pequenos fragmentos. O formato da aplicação é id espaço valor seguido por quebra de linha. Investigue serialização, limites e completude.

Programa C fornecido · abra, leia e copie

A infraestrutura está pronta. Antes de executar, localize as operações destacadas na explicação da prática. O bloco abaixo escreve o fonte e compila com C11, avisos e sem otimização.

Criar e compilar estacao.c
cat > estacao.c <<'FIM'
#define _POSIX_C_SOURCE 200809L
#include <errno.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/wait.h>
#include <unistd.h>

/* Pipe real; texto com id, valor e delimitador. Sem rede. */
static int numero(const char *s, long min, long max) {
    char *fim; errno = 0; long n = strtol(s, &fim, 10);
    if (errno || !*s || *fim || n < min || n > max) return -1;
    return (int)n;
}
static int enviar(int fd, const char *s, size_t n) {
    while (n) {
        ssize_t k = write(fd, s, n);
        if (k < 0 && errno == EINTR) continue;
        if (k <= 0) return -1;
        s += k; n -= (size_t)k;
    }
    return 0;
}
int main(int argc, char **argv) {
    int passo = argc > 1 ? numero(argv[1], 1, 16) : 8;
    int ultimo = argc > 2 ? numero(argv[2], 0, 100) : 30;
    int corte = argc > 3 && strcmp(argv[3], "corte") == 0;
    if (argc > 4 || passo < 0 || ultimo < 0 || (argc > 3 && !corte)) {
        fputs("Uso: ./estacao [bytes:1..16] [ultima:0..100] [corte]\n", stderr);
        return 2;
    }
    signal(SIGPIPE, SIG_IGN);
    alarm(5); /* Limite de contingencia; nao mede timeout distribuido. */
    int canal[2];
    if (pipe(canal) < 0) { perror("pipe"); return 1; }
    pid_t filho = fork();
    if (filho < 0) { perror("fork"); close(canal[0]); close(canal[1]); return 1; }
    if (filho == 0) {
        alarm(5); close(canal[0]);
        int dados[] = {12, 18, 24, ultimo};
        for (int i = 0; i < 4; ++i) {
            char linha[32];
            int n = snprintf(linha, sizeof linha, "%d %d\n", i + 1, dados[i]);
            if (corte && i == 3) --n;
            if (enviar(canal[1], linha, (size_t)n) < 0) _exit(1);
        }
        close(canal[1]); _exit(0);
    }
    close(canal[1]);
    char bloco[16], linha[32]; size_t usados = 0;
    int total = 0, recebidos = 0, falhou = 0, chamadas = 0;
    for (;;) {
        ssize_t n = read(canal[0], bloco, (size_t)passo);
        if (n < 0 && errno == EINTR) continue;
        if (n < 0) { perror("read"); falhou = 1; break; }
        if (!n) break;
        ++chamadas;
        printf("Fragmento: %zd bytes\n", n);
        for (ssize_t i = 0; i < n; ++i) {
            if (bloco[i] != '\n') {
                if (usados + 1 >= sizeof linha) { falhou = 1; break; }
                linha[usados++] = bloco[i];
            } else {
                int id, valor; char extra;
                linha[usados] = '\0';
                if (sscanf(linha, "%d %d %c", &id, &valor, &extra) != 2 ||
                    id != recebidos + 1 || valor < 0 || valor > 100 || recebidos >= 4) {
                    falhou = 1; break;
                }
                ++recebidos; total += valor; usados = 0;
                printf("Mensagem: produtor=P id=%d valor=%d\n", id, valor);
            }
        }
        if (falhou) break;
    }
    close(canal[0]);
    int status;
    while (waitpid(filho, &status, 0) < 0) {
        if (errno != EINTR) { perror("waitpid"); return 1; }
    }
    if (!WIFEXITED(status) || WEXITSTATUS(status)) falhou = 1;
    printf("Chamadas com dados=%d\n", chamadas);
    if (usados) { puts("Registro incompleto: nao incorporado"); falhou = 1; }
    if (recebidos != 4) falhou = 1;
    printf("Leituras=%d Total=%d Media=%.2f Relatorio=%s\n", recebidos, total,
           recebidos ? (double)total / recebidos : 0.0, falhou ? "INCOMPLETO" : "COMPLETO");
    alarm(0); return falhou ? 1 : 0;
}
FIM

Interprete o programa e a saída

enviar repete write se necessário. O laço do pai aceita read parcial, acumula bytes em linha e só interpreta ao encontrar \n. O limite do buffer evita aceitar registros sem tamanho máximo. EOF é fim de fluxo, não um delimitador implícito de mensagem. Fechar todas as pontas de escrita permite observar EOF. O texto evita depender do layout, padding e ordem dos bytes de uma struct C. read bloqueante não transforma a aplicação inteira em um protocolo de requisição e resposta; um pipe também não implementa uma fila persistente.

Mudança orientada

Reduza apenas o limite de leitura para 1 byte. Depois use o caso corte, que retira a quebra de linha do último registro. Preveja o que será aceito e se o relatório poderá ser declarado completo.

Executar mudança orientada
./estacao 1 30
./estacao 1 30 corte
Referência para comparação · abra depois de prever
Sem corte: Leituras=4 Total=84 Media=21.00 Relatorio=COMPLETO
Com corte: Registro incompleto: nao incorporado
Leituras=3 Total=54 Media=18.00 Relatorio=INCOMPLETO
O último comando termina com código 1, intencionalmente.
Limites da conclusão: A fragmentação é provocada pelo limite de read; não mede pacotes, perdas ou latência de rede. TCP é fluxo de bytes e também exige enquadramento na aplicação. UDP preserva fronteiras de datagramas, mas a aplicação precisa tratar truncamento, perdas, duplicatas e ordenação conforme seus requisitos. Não equipare TCP a confirmação de processamento da leitura.

Extensão opcional — comunicação TCP real em loopback

Execute somente se o professor confirmar que sockets locais são permitidos na conta utilizada, ou em WSL/laboratório autorizado. O núcleo em pipe continua completo sem esta extensão. Não altere firewall nem procure outras portas. O programa usa 127.0.0.1, porta atribuída pelo sistema, um filho e limite de cinco segundos. Não escuta nas interfaces da rede.

Programa C fornecido · abra, leia e copie

A infraestrutura está pronta. Antes de executar, localize as operações destacadas na explicação da prática. O bloco abaixo escreve o fonte e compila com C11, avisos e sem otimização.

Criar e compilar tcp_local.c
cat > tcp_local.c <<'FIM'
#define _POSIX_C_SOURCE 200809L
#include <arpa/inet.h>
#include <errno.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <sys/socket.h>
#include <sys/wait.h>
#include <unistd.h>

/* Extensao somente em ambiente autorizado. Loopback; porta efemera. */
static void erro(const char *onde) { perror(onde); exit(1); }
int main(void) {
    signal(SIGPIPE, SIG_IGN); alarm(5);
    int servidor = socket(AF_INET, SOCK_STREAM, 0);
    if (servidor < 0) erro("socket");
    struct sockaddr_in endereco = {0};
    endereco.sin_family = AF_INET;
    endereco.sin_addr.s_addr = htonl(INADDR_LOOPBACK);
    endereco.sin_port = 0;
    if (bind(servidor, (struct sockaddr *)&endereco, sizeof endereco) < 0) erro("bind");
    socklen_t tamanho = sizeof endereco;
    if (getsockname(servidor, (struct sockaddr *)&endereco, &tamanho) < 0) erro("getsockname");
    if (listen(servidor, 1) < 0) erro("listen");
    pid_t filho = fork();
    if (filho < 0) erro("fork");
    if (filho == 0) {
        alarm(5); close(servidor);
        int cliente = socket(AF_INET, SOCK_STREAM, 0);
        if (cliente < 0) _exit(1);
        if (connect(cliente, (struct sockaddr *)&endereco, sizeof endereco) < 0) { perror("connect"); _exit(1); }
        const char texto[] = "12\n18\n24\n30\n";
        size_t enviados = 0;
        while (enviados < sizeof texto - 1) {
            ssize_t n = send(cliente, texto + enviados, sizeof texto - 1 - enviados, 0);
            if (n < 0 && errno == EINTR) continue;
            if (n <= 0) _exit(1);
            enviados += (size_t)n;
        }
        close(cliente); _exit(0);
    }
    int conexao;
    do { conexao = accept(servidor, NULL, NULL); } while (conexao < 0 && errno == EINTR);
    if (conexao < 0) { perror("accept"); close(servidor); (void)waitpid(filho, NULL, 0); return 1; }
    close(servidor);
    char linha[16], c; size_t usados = 0; int total = 0, registros = 0, falhou = 0;
    for (;;) {
        ssize_t n = recv(conexao, &c, 1, 0);
        if (n < 0 && errno == EINTR) continue;
        if (n < 0) { perror("recv"); falhou = 1; break; }
        if (!n) break;
        if (c == '\n') {
            linha[usados] = '\0'; char *fim;
            errno = 0; long valor = strtol(linha, &fim, 10);
            if (errno || !usados || *fim || valor < 0 || valor > 100 || registros >= 4) { falhou = 1; break; }
            total += (int)valor; ++registros; usados = 0;
        } else if (usados + 1 < sizeof linha) linha[usados++] = c;
        else { falhou = 1; break; }
    }
    close(conexao); int status;
    while (waitpid(filho, &status, 0) < 0) if (errno != EINTR) erro("waitpid");
    if (usados || registros != 4 || !WIFEXITED(status) || WEXITSTATUS(status)) falhou = 1;
    printf("TCP loopback: leituras=%d total=%d resultado=%s\n", registros, total, falhou ? "ERRO" : "OK");
    alarm(0); return falhou ? 1 : 0;
}

Se bloqueado, registre o erro ou encerramento por tempo limite e mantenha a investigação em pipe.

O filho conecta, envia quatro linhas e fecha a conexão. O pai aceita uma conexão, recebe um byte por chamada, recompõe linhas e soma. send pode aceitar só parte dos bytes: o laço completa o envio. recv é deliberadamente limitado a um byte; isso não demonstra como o TCP segmentou pacotes. A conexão é unidirecional nos dados da aplicação: o cliente não recebe uma resposta de processamento. O alarme é contingência do programa, não detector de falhas distribuídas. Em caminhos de erro, o sistema fecha descritores ao encerrar o processo; no caminho normal o pai também recolhe o filho.
