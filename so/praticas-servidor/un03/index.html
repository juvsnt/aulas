O cenário do servidor

Cache de páginas, proteção de memória e paginação

Para responder rapidamente aos visitantes sem sobrecarregar o disco rígido ou SSD, o servidor web mantém as páginas e imagens mais acessadas em um buffer na memória RAM. Mas como o sistema operacional organiza esse espaço? Ele cria a ilusão da Memória Virtual: cada processo enxerga um espaço limpo e contíguo, enquanto o hardware divide a memória real em blocos de 4 KB (páginas e molduras/frames). Quando a RAM enche, o algoritmo de substituição precisa decidir inteligentemente qual imagem antiga descartar.

P07 · 60 minutos · Espaço de endereçamento e alocação

O buffer de cache no Heap do servidor

Crie o programa memoria_servidor.c. Ele aloca dinamicamente 1 MB no Heap com malloc() para o cache de páginas e imprime os endereços virtuais das diferentes regiões de memória (Código, Dados, Heap e Stack):

Criar memoria_servidor.c
cat > memoria_servidor.c <<'FIM'
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

int variavel_global = 10; // Segmento de Dados/BSS

int main(void) {
    int variavel_local = 20; // Pilha (Stack)

    size_t tamanho_cache = 1024 * 1024; // 1 MB
    char *cache_paginas = (char *)malloc(tamanho_cache);
    if (!cache_paginas) {
        perror("malloc");
        return 1;
    }

    for (size_t i = 0; i < tamanho_cache; i += 4096) {
        cache_paginas[i] = 'A'; // 1 escrita por pagina de 4 KB
    }

    printf("=== ESPACO DE ENDERECAMENTO DO SERVIDOR (PID: %ld) ===\n", (long)getpid());
    printf("1. Codigo/Texto (funcao main):   %p\n", (void *)main);
    printf("2. Dados Globais:                %p\n", (void *)&variavel_global);
    printf("3. Heap (Buffer de 1MB alocado): %p\n", (void *)cache_paginas);
    printf("4. Pilha/Stack (var local):      %p\n\n", (void *)&variavel_local);

    puts("Memoria alocada com sucesso no Heap para cache do servidor.");
    free(cache_paginas);
    return 0;
}
FIM
gcc -std=c11 -Wall -Wextra -O0 memoria_servidor.c -o memoria_servidor
./memoria_servidor

P08 · 60 minutos · Paginação e memória virtual

Tradução de páginas virtuais e Falta de Página (Page Fault)

Crie o programa traducao_paginas.c. Ele simula a MMU (Unidade de Gerenciamento de Memória) dividindo um endereço virtual em Número de Página e Deslocamento (com páginas de 4 KB). Teste com o endereço 9000 (página presente) e depois com 14000 (página ausente na RAM, causando Page Fault):

Criar traducao_paginas.c
cat > traducao_paginas.c <<'FIM'
#include <stdio.h>
#include <stdlib.h>

#define TAMANHO_PAGINA 4096 // 4 KB por pagina

int main(int argc, char **argv) {
    long endereco_virtual = (argc > 1) ? atol(argv[1]) : 9000;

    printf("=== TRADUCAO DE MEMORIA VIRTUAL NO SERVIDOR ===\n");
    printf("Endereco virtual solicitado pelo processo: %ld\n", endereco_virtual);

    long numero_pagina = endereco_virtual / TAMANHO_PAGINA;
    long deslocamento  = endereco_virtual % TAMANHO_PAGINA;

    printf("Numero da Pagina Virtual (VPN): %ld\n", numero_pagina);
    printf("Deslocamento na Pagina (Offset): %ld bytes\n\n", deslocamento);

    int tabela_frames[] = {5, 8, 2, -1};
    int total_paginas = 4;

    if (numero_pagina < total_paginas) {
        int frame = tabela_frames[numero_pagina];
        if (frame >= 0) {
            long endereco_fisico = (frame * TAMANHO_PAGINA) + deslocamento;
            printf("Status: PAGINA PRESENTE NA RAM!\n");
            printf("Mapeada para o Frame Fisico: %d\n", frame);
            printf("Endereco Fisico Real na RAM: %ld\n", endereco_fisico);
        } else {
            puts("Status: [PAGE FAULT] Pagina ausente da memoria RAM!");
            puts("O kernel precisa suspender o processo e buscar a pagina no disco.");
        }
    } else {
        puts("Status: [SEGMENTATION FAULT] Endereco fora do espaco alocado!");
    }

    return 0;
}
FIM

P09 · 60 minutos · Substituição de páginas

Cache de imagens na RAM: FIFO vs. LRU

Crie o programa cache_paginas.c. O servidor tem espaço para apenas 3 imagens na RAM. Compare a taxa de faltas de página entre o algoritmo FIFO (descarta a que entrou primeiro) e LRU (descarta a que foi usada menos recentemente):

Criar cache_paginas.c
cat > cache_paginas.c <<'FIM'
#include <stdio.h>
#include <string.h>

#define TOTAL_ACESSOS 6
#define MOLDURAS 3

int main(int argc, char **argv) {
    const char *algo = (argc > 1) ? argv[1] : "lru";
    int acessos[TOTAL_ACESSOS] = {1, 2, 3, 1, 4, 2};

    printf("=== CACHE DE IMAGENS DO SERVIDOR: ALGORITMO %s ===\n",
           (strcmp(algo, "fifo") == 0) ? "FIFO (First-In First-Out)" : "LRU (Least Recently Used)");
    printf("Capacidade da memoria RAM: %d imagens em cache\n\n", MOLDURAS);

    int frames[MOLDURAS] = {-1, -1, -1};
    int tempo_uso[MOLDURAS] = {0, 0, 0};
    int ordem_chegada[MOLDURAS] = {0, 0, 0};
    int faltas = 0, hits = 0;
    int relogio = 0;

    for (int i = 0; i < TOTAL_ACESSOS; i++) {
        int pagina = acessos[i];
        relogio++;
        int hit = 0;

        for (int m = 0; m < MOLDURAS; m++) {
            if (frames[m] == pagina) {
                hit = 1;
                tempo_uso[m] = relogio;
                break;
            }
        }

        if (hit) {
            hits++;
            printf("Acesso a Imagem %d: [HIT] Ja estava em cache na RAM!\n", pagina);
        } else {
            faltas++;
            int vitima = 0;
            int encontrou_vazio = 0;
            for (int m = 0; m < MOLDURAS; m++) {
                if (frames[m] == -1) {
                    vitima = m;
                    encontrou_vazio = 1;
                    break;
                }
            }

            if (!encontrou_vazio) {
                if (strcmp(algo, "fifo") == 0) {
                    int menor_chegada = ordem_chegada[0];
                    for (int m = 1; m < MOLDURAS; m++) {
                        if (ordem_chegada[m] < menor_chegada) {
                            menor_chegada = ordem_chegada[m];
                            vitima = m;
                        }
                    }
                } else {
                    int menor_uso = tempo_uso[0];
                    for (int m = 1; m < MOLDURAS; m++) {
                        if (tempo_uso[m] < menor_uso) {
                            menor_uso = tempo_uso[m];
                            vitima = m;
                        }
                    }
                }
            }

            int imagem_expulsa = frames[vitima];
            frames[vitima] = pagina;
            tempo_uso[vitima] = relogio;
            ordem_chegada[vitima] = relogio;

            if (imagem_expulsa != -1) {
                printf("Acesso a Imagem %d: [FALTA DE PAGINA] Expulsou Imagem %d da RAM\n",
                       pagina, imagem_expulsa);
            } else {
                printf("Acesso a Imagem %d: [FALTA DE PAGINA] Carregou no frame livre %d\n",
                       pagina, vitima);
            }
        }
    }

    printf("\n=== RESULTADO DA SUBSTITUICAO (%s) ===\n", algo);
    printf("Total de acessos: %d\n", TOTAL_ACESSOS);
    printf("Faltas de pagina: %d | Hits em RAM: %d\n", faltas, hits);

    return 0;
}
FIM
gcc -std=c11 -Wall -Wextra -O0 cache_paginas.c -o cache_paginas
./cache_paginas lru
./cache_paginas fifo
