O cenário do servidor

Multitarefa, trabalhadores e concorrência

Um servidor web comercial precisa receber dezenas ou centenas de requisições por segundo. Como o sistema operacional organiza esse trabalho? Ele pode criar processos filhos isolados (arquitetura multi-processos), alternar a CPU rapidamente entre requisições usando algoritmos de escalonamento e permitir que múltiplas threads atualizem variáveis compartilhadas (como curtidas e visualizações) com segurança através de travas (Mutex).

P03 · 60 minutos · Controle de processos e sinais

O processo do servidor em segundo plano (Background)

Um servidor não deve travar o terminal do administrador. Ele roda como serviço em segundo plano (background). Vamos simular o controle de um processo via sinais do Linux (STOP para pausar e CONT para retomar):

Execute no terminal
sleep 2 &
pid_servidor=$!
kill -STOP "$pid_servidor"
ps -p "$pid_servidor" -o pid,ppid,stat,comm
kill -CONT "$pid_servidor"
wait "$pid_servidor"
echo "Processo do servidor concluido e recolhido."
Saída típica esperada

...    ... T    sleep

P04 · 60 minutos · Processos trabalhadores com fork

Atendendo clientes com fork, exec e wait

Crie o programa servidor_fork.c. O processo pai (master) recebe a conexão e cria um processo filho com fork() para atender o cliente. O filho substitui sua imagem com execl() e retorna o status HTTP 200:

Criar servidor_fork.c
cat > servidor_fork.c <<'FIM'
#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <sys/wait.h>
#include <string.h>

int main(int argc, char **argv) {
    if (argc == 2 && strcmp(argv[1], "trabalhador") == 0) {
        printf("[PROCESSO FILHO APOS EXEC] PID = %ld | PPID = %ld\n",
               (long)getpid(), (long)getppid());
        puts("Trabalhador processou o pedido do cliente com sucesso.");
        return 200; // Codigo HTTP 200 OK retornado como exit status
    }

    printf("=== SERVIDOR MASTER (PAI): GERENCIANDO REQUISICOES ===\n");
    int conexoes_ativas = 1;
    printf("Servidor Pai iniciado: PID = %ld | Conexoes = %d\n",
           (long)getpid(), conexoes_ativas);

    puts("\n[CLIENTE CONECTOU] Criando processo filho (fork) para atender o cliente...");
    fflush(stdout);

    pid_t pid = fork();

    if (pid < 0) {
        perror("fork");
        return 1;
    }

    if (pid == 0) {
        conexoes_ativas = 99; // Altera apenas a memoria privada do filho
        printf("[FILHO ANTES DO EXEC] PID = %ld | PPID = %ld | Minhas conexoes = %d\n",
               (long)getpid(), (long)getppid(), conexoes_ativas);
        fflush(stdout);

        execl("./servidor_fork", "servidor_fork", "trabalhador", (char *)NULL);
        perror("execl");
        _exit(1);
    }

    int status;
    waitpid(pid, &status, 0);

    puts("\n=== RESUMO NO SERVIDOR PAI ===");
    printf("Conexoes no Pai continuam intactas: %d (Memoria isolada do filho!)\n", conexoes_ativas);
    if (WIFEXITED(status)) {
        printf("Processo filho terminou com status HTTP = %d\n", WEXITSTATUS(status));
    }

    return 0;
}
FIM
gcc -std=c11 -Wall -Wextra -O0 servidor_fork.c -o servidor_fork
./servidor_fork

P05 · 60 minutos · Escalonamento da CPU

Fila de requisições: FCFS, SJF e Round Robin

Crie o programa escalonador_web.c para comparar como o sistema operacional decide a ordem de atendimento entre um download longo (5s) e um clique leve (1s):

Criar escalonador_web.c
cat > escalonador_web.c <<'FIM'
#include <stdio.h>
#include <string.h>

int main(int argc, char **argv) {
    const char *politica = (argc > 1) ? argv[1] : "fcfs";

    // 3 Requisicoes chegam no instante t=0:
    // Req 1: Download de video longo (burst = 5s)
    // Req 2: Download de imagem media (burst = 3s)
    // Req 3: Clique leve de texto     (burst = 1s)

    printf("=== ESCALONAMENTO DE REQUISICOES NO SERVIDOR (%s) ===\n", politica);

    if (strcmp(politica, "fcfs") == 0) {
        puts("Ordem: 1. Video (5s) -> 2. Imagem (3s) -> 3. Texto (1s)");
        int espera1 = 0;
        int espera2 = 5;
        int espera3 = 5 + 3; // 8s de espera
        double media = (espera1 + espera2 + espera3) / 3.0;

        puts("\nDiagrama de Gantt:");
        puts("[Req1: 0 a 5s] [Req2: 5 a 8s] [Req3: 8 a 9s]");
        printf("Tempo de espera: Req1=%ds | Req2=%ds | Req3=%ds\n", espera1, espera2, espera3);
        printf("Tempo medio de espera: %.2f segundos\n", media);
        puts("Problema do FCFS (Efeito Comboio): O clique rapido esperou 8s pelo video longo!");

    } else if (strcmp(politica, "sjf") == 0) {
        puts("Ordem: 1. Texto (1s) -> 2. Imagem (3s) -> 3. Video (5s)");
        int espera3 = 0;
        int espera2 = 1;
        int espera1 = 1 + 3; // 4s de espera
        double media = (espera3 + espera2 + espera1) / 3.0;

        puts("\nDiagrama de Gantt:");
        puts("[Req3: 0 a 1s] [Req2: 1 a 4s] [Req1: 4 a 9s]");
        printf("Tempo de espera: Req3=%ds | Req2=%ds | Req1=%ds\n", espera3, espera2, espera1);
        printf("Tempo medio de espera: %.2f segundos\n", media);
        puts("Vantagem do SJF: Menor tempo medio de espera possivel!");

    } else if (strcmp(politica, "rr") == 0) {
        puts("Modo Round Robin (fatias de tempo de 2s):");
        puts("[Req1(2s)] [Req2(2s)] [Req3(1s: Fim)] [Req1(2s)] [Req2(1s: Fim)] [Req1(1s: Fim)]");
        puts("Tempo de resposta imediato para todos os clientes sem travamentos!");
    }

    return 0;
}
FIM
gcc -std=c11 -Wall -Wextra -O0 escalonador_web.c -o escalonador_web
./escalonador_web fcfs
./escalonador_web sjf
./escalonador_web rr

P06 · 60 minutos · Concorrência e exclusão mútua

O contador de curtidas (Likes) e o Mutex

Crie o programa contador_likes.c. Ele cria 2 threads simultâneas simulando visitantes clicando em "Curtir" 50.000 vezes cada. Execute primeiro sem mutex (veja as curtidas se perdendo por condição de corrida) e depois com mutex:

Criar contador_likes.c
cat > contador_likes.c <<'FIM'
#include <stdio.h>
#include <stdlib.h>
#include <pthread.h>
#include <string.h>

#define CLICKS_POR_CLIENTE 50000

long contador_likes = 0;
pthread_mutex_t trava_likes = PTHREAD_MUTEX_INITIALIZER;
int usar_mutex = 0;

void *cliente_clicando(void *arg) {
    (void)arg;
    for (int i = 0; i < CLICKS_POR_CLIENTE; i++) {
        if (usar_mutex) {
            pthread_mutex_lock(&trava_likes);
            contador_likes++; // Regiao critica protegida
            pthread_mutex_unlock(&trava_likes);
        } else {
            long temp = contador_likes;
            for (volatile int j = 0; j < 5; j++);
            contador_likes = temp + 1;
        }
    }
    return NULL;
}

int main(int argc, char **argv) {
    usar_mutex = (argc > 1 && strcmp(argv[1], "com_mutex") == 0);

    printf("=== SERVIDOR WEB: 2 CLIENTES CLICANDO EM CURTIR (%s) ===\n",
           usar_mutex ? "COM MUTEX" : "SEM MUTEX (RACE CONDITION)");
    printf("Total esperado: 2 clientes x %d cliques = %d curtidas\n",
           CLICKS_POR_CLIENTE, CLICKS_POR_CLIENTE * 2);

    pthread_t thread1, thread2;
    pthread_create(&thread1, NULL, cliente_clicando, NULL);
    pthread_create(&thread2, NULL, cliente_clicando, NULL);

    pthread_join(thread1, NULL);
    pthread_join(thread2, NULL);

    printf("Total REAL computado no servidor: %ld curtidas\n", contador_likes);

    if (contador_likes == CLICKS_POR_CLIENTE * 2) {
        puts("Sucesso! Mutex garantiu que nenhum clique fosse perdido.");
    } else {
        printf("Perda detectada! Faltaram %ld curtidas por falta de exclusao mutua.\n",
               (CLICKS_POR_CLIENTE * 2) - contador_likes);
    }

    return 0;
}
FIM
gcc -std=c11 -Wall -Wextra -O0 -pthread contador_likes.c -o contador_likes
./contador_likes sem_mutex
./contador_likes com_mutex
Trecho esperado na execução com e sem mutex
SEM MUTEX: Total REAL computado: ... (Perda detectada!)
COM MUTEX: Total REAL computado: 100000 curtidas (Sucesso!)
