02Paralelismo e arquiteturas não convencionais

Conte recursos, reconheça a quota

Execute no terminal
LC_ALL=C lscpu | head -n 18
sed -n '/Cpus_allowed_list/p' /proc/self/status

O primeiro comando descreve o processador; o segundo mostra a lista de CPUs em que o processo pode ser escalonado, conforme /proc/self/status. Essa afinidade não equivale à quota de tempo de CPU. A documentação do LIAC estabelece limite de uma CPU por conta: dois processos podem compartilhar essa quota, mesmo que o Linux mostre oito processadores lógicos.

Não há necessidade de uma GPU para o exercício. Esta prática não verifica disponibilidade de GPU, não usa CUDA e não conecta máquinas pela rede.
03Paralelismo e arquiteturas não convencionais

Separe o lote em duas partes

A estação terá dois trabalhadores locais: um soma 12 e 18, o outro soma 24 e 30. Cada um repete a soma 10 mil vezes para formar uma tarefa pequena e finita. Antes de executar, preveja as duas somas parciais e o total acumulado.

Cole o bloco inteiro. A primeira linha inicia a gravação; a linha final FIM, sozinha, encerra. O arquivo fica nesta pasta. O símbolo > no prompt durante a colagem significa que o shell está esperando o fim do bloco.

Criar paralelo.c
cat > paralelo.c <<'FIM'
#define _POSIX_C_SOURCE 200809L
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/wait.h>
#include <time.h>
#include <errno.h>

/* Infraestrutura POSIX/Linux fornecida: foco na divisao e na reducao. */
static long trabalho(const int *valores, int n) {
    long total = 0;
    for (int repeticao = 0; repeticao < 10000; repeticao++)
        for (int i = 0; i < n; i++) total += valores[i];
    return total;
}
static double agora(void) {
    struct timespec t;
    if (clock_gettime(CLOCK_MONOTONIC, &t)) { perror("clock_gettime"); exit(1); }
    return t.tv_sec + t.tv_nsec / 1e9;
}
/* Transfere todos os bytes, tratando interrupcoes e transferencias parciais. */
static int transferir(int fd, void *dado, size_t n, int escrever) {
    unsigned char *p = dado;
    while (n) {
        ssize_t k = escrever ? write(fd, p, n) : read(fd, p, n);
        if (k < 0 && errno == EINTR) continue;
        if (k <= 0) return 0;
        p += k; n -= (size_t)k;
    }
    return 1;
}
static int esperar(pid_t pid) {
    int status;
    while (waitpid(pid, &status, 0) < 0) {
        if (errno != EINTR) return 0;
    }
    return WIFEXITED(status) && WEXITSTATUS(status) == 0;
}
int main(int argc, char **argv) {
    if (argc == 2 && !strcmp(argv[1], "amdahl")) {
        double p = 0.8, n = 4;
        printf("%.1f\n", 1.0 / ((1-p) + p/n)); return 0;
    }
    if (argc > 1) {
        if (argc != 3) { fputs("Use dois valores ou amdahl.\n", stderr); return 1; }
        int valores[2];
        for (int i = 0; i < 2; i++) {
            char *fim;
            long v = strtol(argv[i+1], &fim, 10);
            if (fim == argv[i+1] || *fim || v < -1000 || v > 1000) return 1;
            valores[i] = (int)v;
        }
        printf("%ld\n", trabalho(valores, 2)); return 0;
    }
    int valores[] = {12,18,24,30};
    double inicio = agora();
    long serial = trabalho(valores, 4);
    double ts = agora() - inicio;
    inicio = agora();
    int canal[2];
    pid_t filhos[2];
    if (pipe(canal)) { perror("pipe"); return 1; }
    int criados = 0;
    for (int i = 0; i < 2; i++) {
        pid_t pid = fork();
        if (pid < 0) {
            perror("fork: avise o professor");
            close(canal[0]); close(canal[1]);
            for (int j = 0; j < criados; j++) esperar(filhos[j]);
            return 1;
        }
        if (pid == 0) {
            close(canal[0]);
            long parcial = trabalho(valores + 2*i, 2);
            int ok = transferir(canal[1], &parcial, sizeof parcial, 1);
            close(canal[1]); _exit(ok ? 0 : 1);
        }
        filhos[criados++] = pid;
    }
    close(canal[1]);
    long dividido = 0;
    int ok = 1;
    for (int i = 0; i < 2; i++) {
        long parcial;
        if (!transferir(canal[0], &parcial, sizeof parcial, 0)) { ok = 0; break; }
        dividido += parcial; /* Coordenador reduz as duas somas parciais. */
    }
    close(canal[0]);
    for (int i = 0; i < 2; i++) if (!esperar(filhos[i])) ok = 0;
    if (!ok) { fputs("Falha nos filhos; avise o professor.\n", stderr); return 1; }
    double tp = agora() - inicio;
    printf("Serial=%ld Dividido=%ld Iguais=%s\n", serial, dividido, serial == dividido ? "verdadeiro" : "falso");
    printf("t_serial=%.9fs t_dividido=%.9fs\n", ts, tp);
    if (tp > 0) printf("Speedup observado=%.3f\n", ts/tp);
    else puts("Speedup nao calculavel: tempo abaixo da resolucao do relogio");
    printf("Amdahl p=0.5, N=2: %.16f\n", 1.0 / (0.5 + 0.5/2));
    return 0;
}
FIM

trabalho soma os valores em 10 mil repetições. fork cria cada filho com seu próprio espaço de memória; pipe fornece um canal local para as parciais; waitpid aguarda a conclusão. O pai soma os dois resultados recebidos. A função transferir trata leituras e escritas parciais e interrupções. Essa infraestrutura é fornecida: acompanhe primeiro divisão, soma parcial e redução; depois relacione os recursos do Linux às funções. Argumentos numéricos executam somente o trabalho local; amdahl calcula o exemplo teórico.

04Paralelismo e arquiteturas não convencionais

Execute e verifique antes de cronometrar

Execute no terminal
gcc -std=c11 -O0 -Wall -Wextra -Wpedantic paralelo.c -o paralelo
./paralelo
Resultados fixos e campos de tempo variáveis
Serial=840000 Dividido=840000 Iguais=verdadeiro
t_serial=...s t_dividido=...s
Speedup observado=...
Amdahl p=0.5, N=2: 1.3333333333333333

O coordenador faz a redução: soma as duas parciais. 840000 é o total bruto 84 repetido 10 mil vezes, não uma nova temperatura. O caso dividido inclui criar os processos e coletar resultados; o serial não tem esse custo. Dois filhos existentes não provam execução simultânea em dois núcleos.

No servidor compartilhado, valores de tempo variam. Para esta tarefa curta, a versão dividida provavelmente será mais lenta, mas isso não é obrigatório. Se um filho for impedido pela política da conta, registre a restrição e faça a redução no papel; não altere os limites.

Saída esperada · dados fixos
300000
540000

Na organização, o coordenador recebe resultados de processos com memória separada: podemos relacioná-la a múltiplos fluxos de instruções e dados (MIMD), sem concluir que houve paralelismo físico. Aplicar a mesma adição simultaneamente a vários elementos em uma instrução vetorial seria SIMD. Um vetor em C, por si só, não demonstra SIMD de hardware.

06Paralelismo e arquiteturas não convencionais

Calcule o limite e feche o projeto

Com fração paralelizável p e N processadores, Amdahl idealiza a aceleração como 1 / ((1-p) + p/N). Para p=0,5, dois processadores dão 1,33; mesmo com infinitos, o teto é 2. Comunicação, criação de processos e quotas podem reduzir o ganho observado.

Para o projeto da estação, uma GPU ou acelerador poderia ser útil para muitos sensores ou imagens, mas o custo de transferência e o volume precisam justificar a escolha. Arquiteturas heterogêneas combinam unidades especializadas; computação neuromórfica, processamento próximo à memória e computação quântica não são recursos testados por este exercício.

Execute no terminal
./paralelo amdahl
Saída esperada · dados fixos
2.5

Conclua a história: bits representam leituras; o compilador gera instruções; a CPU move e calcula; a hierarquia traz dados; E/S recebe e publica; pipeline sobrepõe estágios; processos dividem trabalho. Nenhuma dessas escolhas melhora o sistema isoladamente em todas as situações.
