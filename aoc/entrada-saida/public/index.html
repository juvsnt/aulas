Atividade 1 · até 30 min Medição real

Quem está chamando a CPU?

Objetivo: descobrir quais dispositivos do computador do LIAC interrompem a CPU e com que frequência.

Conceitos

Periférico (peripheral)
O aparelho em si: teclado, disco, placa de rede. slide 4
Interface (interface)
O padrão de ligação física e elétrica do periférico com a placa-mãe, como SATA, USB ou PCIe. slide 4
Controlador (controller)
Chip que comanda o periférico e conversa com a CPU. A CPU diz “leia o bloco X”; o controlador faz o resto. slide 4
Interrupção (interrupt) e IRQ (interrupt request)
Interrupção é o aviso que o controlador manda à CPU: “preciso de atenção”. Cada fonte de aviso tem uma linha numerada, a IRQ. slide 9 e slide 10

O que fazer

  1. Veja os discos do computador. lsblk lista os discos; a coluna TRAN mostra a interface. Se o título das colunas aparecer quebrado em duas linhas, tudo bem.
    Cole no terminal
    cd ~/atividades/aoc-entrada-saida
lsblk -d -o NAME,TRAN,SIZE
# fim do bloco
  2. Pergunta extra. O lsblk mostra o tamanho em GiB (1 GiB = 1.073.741.824 bytes), mas o fabricante vende o disco em TB (1 TB = 1.000.000.000.000 bytes), como kB e KiB na Unidade 4.
  3. Crie o programa interrupcoes. Ele lê o arquivo /proc/interrupts, em que o Linux conta os avisos de cada dispositivo, espera alguns segundos, lê de novo e mostra quem mais cresceu.
    Criar interrupcoes.c
    Cole no terminal para criar interrupcoes.c
    cat > interrupcoes.c <<'FIM'
/* interrupcoes.c - AOC, Unidade 5, Atividade 1.
   Conta quantas interrupcoes cada dispositivo gerou durante alguns segundos.
   O Linux mantem esses contadores em /proc/interrupts: uma linha por IRQ,
   uma coluna por CPU. O programa le o arquivo duas vezes, calcula a
   diferenca e mostra as linhas que mais cresceram. So faz leitura. */
#define _POSIX_C_SOURCE 200809L
#include <ctype.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

#ifndef ARQUIVO
#define ARQUIVO "/proc/interrupts"
#endif
#define MAX_LINHAS 512

typedef struct {
    char nome[24];             /* numero da IRQ (1, 132...) ou sigla (LOC) */
    char dispositivo[64];      /* quem gera a interrupcao */
    unsigned long long total;  /* soma de todas as CPUs */
} Linha;

static char *pula_espacos(char *p) {
    while (isspace((unsigned char)*p)) p++;
    return p;
}

static char *pula_palavra(char *p) {
    p = pula_espacos(p);
    while (*p != '\0' && !isspace((unsigned char)*p)) p++;
    return p;
}

static int ler_contadores(Linha *v) {
    FILE *f = fopen(ARQUIVO, "r");
    if (f == NULL) { perror(ARQUIVO); exit(1); }
    char buf[8192];
    int n = 0, ncpu = 0;
    /* 1a linha: CPU0 CPU1 ... -> conta quantas CPUs existem */
    if (fgets(buf, sizeof buf, f) != NULL)
        for (char *t = strtok(buf, " \t\n"); t != NULL; t = strtok(NULL, " \t\n"))
            ncpu++;
    while (n < MAX_LINHAS && fgets(buf, sizeof buf, f) != NULL) {
        char *p = pula_espacos(buf);
        char *dois_pontos = strchr(p, ':');
        if (dois_pontos == NULL) continue;
        *dois_pontos = '\0';
        snprintf(v[n].nome, sizeof v[n].nome, "%.23s", p);
        p = dois_pontos + 1;
        v[n].total = 0;
        for (int c = 0; c < ncpu; c++) {          /* soma as colunas */
            char *fim;
            unsigned long long x = strtoull(p, &fim, 10);
            if (fim == p) break;
            v[n].total += x;
            p = fim;
        }
        p[strcspn(p, "\n")] = '\0';
        /* "IR-IO-APIC 1-edge i8042": pula controlador e tipo de sinal e
           fica com o dispositivo. Linhas com sigla (LOC) ja trazem a descricao. */
        if (isdigit((unsigned char)v[n].nome[0]))
            p = pula_palavra(pula_palavra(p));
        snprintf(v[n].dispositivo, sizeof v[n].dispositivo, "%.30s", pula_espacos(p));
        n++;
    }
    fclose(f);
    return n;
}

int main(int argc, char **argv) {
    int segundos = argc > 1 ? atoi(argv[1]) : 10;
    if (segundos < 1 || segundos > 60) {
        fprintf(stderr, "uso: %s [segundos de 1 a 60]\n", argv[0]);
        return 1;
    }
    static Linha antes[MAX_LINHAS], depois[MAX_LINHAS];
    unsigned long long cresceu[MAX_LINHAS] = {0}, soma = 0;

    int na = ler_contadores(antes);
    printf("Contando interrupcoes por %d s...\n", segundos);
    fflush(stdout);
    sleep((unsigned)segundos);
    int nd = ler_contadores(depois);

    for (int i = 0; i < nd; i++)
        for (int j = 0; j < na; j++)
            if (strcmp(antes[j].nome, depois[i].nome) == 0) {
                if (depois[i].total > antes[j].total)
                    cresceu[i] = depois[i].total - antes[j].total;
                soma += cresceu[i];
                break;
            }

    printf("\n%-6s %-30s %12s %10s\n", "linha", "dispositivo/evento", "no periodo", "por seg");
    for (int k = 0; k < 8; k++) {                 /* as 8 que mais cresceram */
        int m = -1;
        for (int i = 0; i < nd; i++)
            if (cresceu[i] > 0 && (m < 0 || cresceu[i] > cresceu[m])) m = i;
        if (m < 0) break;
        printf("%-6s %-30s %12llu %10.1f\n", depois[m].nome, depois[m].dispositivo,
               cresceu[m], (double)cresceu[m] / segundos);
        cresceu[m] = 0;
    }
    printf("\nTotal em %d s: %llu interrupcoes (%.0f por segundo)\n",
           segundos, soma, (double)soma / segundos);
    return 0;
}

Atividade 2 · até 30 min Medição real · analogia

Perguntar ou ser avisado?

Objetivo: medir quanto tempo de CPU se gasta esperando um dado de duas formas diferentes.

Conceitos

E/S programada (programmed I/O)
A CPU comanda a operação e acompanha cada passo sozinha. slide 8
Espera ocupada (busy-waiting) ou polling
A CPU pergunta sem parar “já terminou?” e não faz mais nada enquanto isso. slide 8
E/S por interrupção (interrupt-driven I/O)
A CPU pede e vai fazer outra coisa; o dispositivo avisa quando terminar. slide 9
Tempo de CPU (CPU time)
Quanto tempo o processador trabalhou para o programa. Diferente do tempo de espera, que conta o relógio do início ao fim.

Analogia: aqui o “dispositivo” é outro programa (um processo filho, criado pelo próprio programa), que manda 1 byte depois de 3 s por um pipe, um canal entre programas. A medição é real, mas não há periférico de verdade. No modo avisado, quem acorda o programa é o sistema operacional.

O que fazer

  1. Crie e compile o programa:
    Criar espera.c
    Cole no terminal para criar espera.c
    cat > espera.c <<'FIM'
/* espera.c - AOC, Unidade 5, Atividade 2.
   Duas formas de esperar um dado que so fica pronto depois de 3 segundos.
     perguntando: consulta sem parar se o dado chegou (como o polling).
     avisado:     dorme; o sistema operacional acorda o programa quando o
                  dado chega (como a interrupcao).
   O "dispositivo" e um processo filho que envia 1 byte por um pipe.
   E uma analogia em software: nenhum periferico real e usado. */
#define _POSIX_C_SOURCE 200809L
#include <errno.h>
#include <fcntl.h>
#include <stdio.h>
#include <string.h>
#include <sys/resource.h>
#include <sys/wait.h>
#include <time.h>
#include <unistd.h>

static double relogio(void) {          /* tempo total decorrido, em s */
    struct timespec t;
    clock_gettime(CLOCK_MONOTONIC, &t);
    return (double)t.tv_sec + (double)t.tv_nsec / 1e9;
}

static double tempo_cpu(void) {        /* tempo de CPU deste programa, em s */
    struct rusage r;
    getrusage(RUSAGE_SELF, &r);
    return (double)(r.ru_utime.tv_sec + r.ru_stime.tv_sec)
         + (double)(r.ru_utime.tv_usec + r.ru_stime.tv_usec) / 1e6;
}

int main(int argc, char **argv) {
    int perguntando = argc > 1 && strcmp(argv[1], "perguntando") == 0;
    int avisado = argc > 1 && strcmp(argv[1], "avisado") == 0;
    if (!perguntando && !avisado) {
        fprintf(stderr, "uso: %s perguntando | avisado\n", argv[0]);
        return 1;
    }
    int canal[2];
    if (pipe(canal) != 0) { perror("pipe"); return 1; }
    pid_t filho = fork();
    if (filho < 0) { perror("fork"); return 1; }
    if (filho == 0) {                  /* o "dispositivo" demora 3 s */
        close(canal[0]);
        sleep(3);
        if (write(canal[1], "D", 1) != 1) _exit(1);
        _exit(0);
    }
    close(canal[1]);

    double t0 = relogio(), c0 = tempo_cpu();
    unsigned long long consultas = 0;
    char dado = '?';
    if (perguntando) {
        fcntl(canal[0], F_SETFL, O_NONBLOCK);   /* read responde na hora */
        for (;;) {                               /* "Ja terminou?" */
            consultas++;
            ssize_t r = read(canal[0], &dado, 1);
            if (r == 1) break;
            if (r < 0 && errno != EAGAIN) { perror("read"); return 1; }
        }
    } else {
        consultas = 1;                           /* pede uma vez e dorme */
        if (read(canal[0], &dado, 1) != 1) { perror("read"); return 1; }
    }
    double espera = relogio() - t0, cpu = tempo_cpu() - c0;
    waitpid(filho, NULL, 0);

    printf("modo:                 %s\n", argv[1]);
    printf("dado recebido:        %c\n", dado);
    printf("tempo de espera:      %.2f s\n", espera);
    printf("tempo de CPU gasto:   %.3f s (%.0f%% da espera)\n", cpu, 100.0 * cpu / espera);
    printf("consultas feitas:     %llu\n", consultas);
    return 0;
}

Atividade 3 · até 30 min Medição real · analogia

Um byte por vez ou em blocos?

Objetivo: medir quanto custa ler os mesmos 4 MiB com pedidos pequenos e com pedidos grandes.

Conceitos

Sobrecarga (overhead)
Custo fixo de cada pedido, quer ele traga 1 byte ou milhares. No slide, “1 byte = 1 interrupção”. slide 11
Transferência em bloco (block transfer)
Mover muitos bytes de uma só vez, com um único pedido.
DMA (direct memory access, acesso direto à memória)
Controlador que copia blocos inteiros entre o periférico e a RAM sem a CPU. Ela só configura a cópia e recebe uma interrupção no fim. slide 12
MiB e ms
1 MiB = 1.048.576 bytes. 1 ms é um milésimo de segundo; 1 ns, um bilionésimo.

Analogia: cada pedido aqui é uma chamada ao sistema operacional (read), e os dados vêm da memória, não do disco. O programa mede o custo de cada pedido, não o DMA. O DMA entra no raciocínio: é o hardware que evita esse custo byte a byte nos dispositivos rápidos.

O que fazer

  1. Crie e compile o programa. Ele cria um arquivo de 4 MiB na sua pasta, lê esse arquivo cinco vezes e apaga o arquivo no fim.
    Criar blocos.c
    Cole no terminal para criar blocos.c
    cat > blocos.c <<'FIM'
/* blocos.c - AOC, Unidade 5, Atividade 3.
   Le o mesmo arquivo de 4 MiB cinco vezes, com pedidos (blocos) de
   tamanhos diferentes. Cada read() e um pedido ao sistema operacional.
   Mostra quantos pedidos foram feitos e quanto tempo levou.
   Limite: os dados vem da memoria (cache de arquivos do Linux), nao do
   disco. Mede-se o custo de cada pedido, nao o DMA. */
#define _POSIX_C_SOURCE 200809L
#include <fcntl.h>
#include <stdio.h>
#include <time.h>
#include <unistd.h>

#define TAMANHO (4u * 1024u * 1024u)   /* 4 MiB = 4.194.304 bytes */
#define NOME "dados_4MiB.bin"

static double relogio(void) {
    struct timespec t;
    clock_gettime(CLOCK_MONOTONIC, &t);
    return (double)t.tv_sec + (double)t.tv_nsec / 1e9;
}

int main(void) {
    static unsigned char buf[65536];

    /* 1. cria o arquivo de 4 MiB na pasta atual */
    int fd = open(NOME, O_WRONLY | O_CREAT | O_TRUNC, 0600);
    if (fd < 0) { perror(NOME); return 1; }
    for (unsigned i = 0; i < sizeof buf; i++) buf[i] = (unsigned char)(i * 7u);
    for (unsigned feito = 0; feito < TAMANHO; feito += (unsigned)sizeof buf)
        if (write(fd, buf, sizeof buf) != (ssize_t)sizeof buf) { perror("write"); return 1; }
    close(fd);

    /* 2. le o arquivo inteiro com cada tamanho de bloco */
    const size_t blocos[] = {1, 16, 512, 4096, 65536};
    printf("Arquivo de 4 MiB = %u bytes\n\n", TAMANHO);
    printf("%12s %12s %12s %16s\n", "bloco(bytes)", "pedidos", "tempo(ms)", "ns por pedido");
    for (size_t k = 0; k < sizeof blocos / sizeof blocos[0]; k++) {
        fd = open(NOME, O_RDONLY);
        if (fd < 0) { perror(NOME); return 1; }
        unsigned long long pedidos = 0, lidos = 0;
        ssize_t r;
        double t0 = relogio();
        while ((r = read(fd, buf, blocos[k])) > 0) {
            pedidos++;
            lidos += (unsigned long long)r;
        }
        double ms = (relogio() - t0) * 1e3;
        close(fd);
        if (r < 0 || lidos != TAMANHO) { fprintf(stderr, "erro de leitura\n"); return 1; }
        printf("%12zu %12llu %12.1f %16.1f\n", blocos[k], pedidos, ms,
               ms * 1e6 / (double)pedidos);
    }

    /* 3. apaga o arquivo para nao ocupar a sua cota */
    unlink(NOME);
    return 0;
}

Erro ao compilar: crie o arquivo de novo, copiando o bloco inteiro.

“Permissão negada” ou “arquivo não encontrado”: rode cd ~/atividades/aoc-entrada-saida e tente outra vez. Se continuar, registre a mensagem e avise o professor. Não use sudo.
