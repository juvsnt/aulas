Atividade 1 · até 30 min Simulação

O lava-jato das instruções

Objetivo: ler um diagrama temporal e calcular quanto o pipeline ganha, e quanto não ganha.

Conceitos

Pipeline
Linha de montagem de instruções: a execução é dividida em etapas, e cada etapa trabalha numa instrução diferente ao mesmo tempo. slide 3
Estágio (stage)
Cada etapa da linha. No pipeline clássico são cinco: IF (busca), ID (decodificação), EX (execução), MEM (memória) e WB (escrita do resultado). No livro: B, D, E, M e W. slide 4
Ciclo de relógio (clock cycle)
O “tique” que faz cada instrução avançar um estágio. Se a frequência é f, o ciclo dura t = 1/f. O ciclo precisa caber o estágio mais lento.
Latência × vazão (latency × throughput)
Latência: quanto tempo uma instrução leva do início ao fim. Vazão: quantas instruções terminam por unidade de tempo. slide 6

Simulação: o programa desta atividade não mede o processador. Ele aplica as regras de um pipeline didático de 5 estágios, com ciclo de 2 ns, e desenha o resultado, como o diagrama do slide 5.

O que fazer

  1. Comece pelo lava-jato. Lavar um carro tem 4 etapas de 15 min cada (1 h por carro). Sem pipeline, 5 carros levam 5 h. Com pipeline, entra um carro novo assim que a primeira etapa fica livre.
  2. Crie e compile o simulador. A opção -O1 pede ao compilador uma otimização leve, a mesma usada nas outras atividades.
    Criar pipeline.c
    Cole no terminal para criar pipeline.c
    cat > pipeline.c <<'FIM'
/* pipeline.c - SIMULACAO de um pipeline didatico de 5 estagios.
   Desenha o diagrama temporal (qual instrucao esta em qual estagio a cada
   ciclo) e calcula tempos. Nao mede o processador: so aplica as regras.
   Uso: ./pipeline 5            5 instrucoes sem dependencia
        ./pipeline 1000000      um milhao de instrucoes (so os totais)
        ./pipeline dependencia  ADD, SUB e OR do slide 9, sem adiantamento
        ./pipeline adiantamento as mesmas, com adiantamento (forwarding) */
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define K 5        /* estagios: IF ID EX MEM WB */
#define T_NS 2.0   /* duracao do ciclo de relogio, em nanossegundos */

/* Cada instrucao: texto, registrador escrito e dois registradores lidos. */
struct instr { const char *texto, *escreve, *le1, *le2; };

static const struct instr exemplo[] = {
    {"ADD $s0,$t0,$t1", "$s0", "$t0", "$t1"},
    {"SUB $t2,$s0,$t3", "$t2", "$s0", "$t3"},   /* le $s0: depende do ADD */
    {"OR  $t4,$t5,$t6", "$t4", "$t5", "$t6"},   /* nao depende de ninguem */
};

static void totais(long m, long ciclos, long bolhas) {
    double com = (double)ciclos * T_NS, sem = (double)K * (double)m * T_NS;
    printf("Instrucoes (M):               %ld\n", m);
    printf("Ciclos com pipeline:          %ld\n", ciclos);
    printf("Bolhas (ciclos perdidos):     %ld\n", bolhas);
    printf("Tempo com pipeline:           %.0f ns\n", com);
    printf("Tempo sem pipeline (K*M*t):   %.0f ns\n", sem);
    printf("Latencia sem bolhas (K*t):    %.0f ns por instrucao\n", K * T_NS);
    printf("Ganho (speedup) sem/com:      %.2f vezes\n", sem / com);
    printf("CPI (ciclos por instrucao):   %.2f\n", (double)ciclos / (double)m);
}

static void diagrama(int n, const char **nome, const int *busca, const int *decod) {
    static const char *est[] = {"ID", "EX", "MEM", "WB"};
    int fim = decod[n - 1] + 3;
    printf("%-17s", "ciclo:");
    for (int c = 1; c <= fim; c++) printf("%-4d", c);
    printf("\n");
    for (int i = 0; i < n; i++) {
        printf("%-17s", nome[i]);
        for (int c = 1; c <= fim; c++) {
            const char *s = "";
            if (c == busca[i]) s = "IF";
            else if (c > busca[i] && c < decod[i]) s = "--";   /* bolha */
            else if (c >= decod[i] && c <= decod[i] + 3) s = est[c - decod[i]];
            printf("%-4s", s);
        }
        printf("\n");
    }
}

int main(int argc, char **argv) {
    if (argc != 2) {
        fprintf(stderr, "uso: %s <numero> | dependencia | adiantamento\n", argv[0]);
        return 1;
    }
    printf("SIMULACAO: pipeline de %d estagios, ciclo de %.0f ns\n\n", K, T_NS);
    int dep = strcmp(argv[1], "dependencia") == 0;
    int adi = strcmp(argv[1], "adiantamento") == 0;
    if (dep || adi) {
        int n = 3, busca[3], decod[3];
        const char *nome[3];
        for (int i = 0; i < n; i++) {
            nome[i] = exemplo[i].texto;
            busca[i] = i == 0 ? 1 : decod[i - 1];   /* busca quando a anterior decodifica */
            decod[i] = busca[i] + 1;
            for (int j = 0; j < i && dep; j++)      /* sem adiantamento: espera o WB */
                if (!strcmp(exemplo[j].escreve, exemplo[i].le1) ||
                    !strcmp(exemplo[j].escreve, exemplo[i].le2))
                    if (decod[i] < decod[j] + 3) decod[i] = decod[j] + 3;
        }
        diagrama(n, nome, busca, decod);
        printf("\n-- = bolha (stall): a instrucao espera; um estagio fica sem trabalho util\n\n");
        long ciclos = decod[n - 1] + 3;
        totais(n, ciclos, ciclos - (K + (n - 1)));
        return 0;
    }
    long m = strtol(argv[1], NULL, 10);
    if (m < 1 || m > 1000000000L) {
        fprintf(stderr, "use um numero de 1 a 1000000000\n");
        return 1;
    }
    if (m <= 8) {
        int busca[8], decod[8];
        const char *nome[8];
        static const char *rot[] = {"I1", "I2", "I3", "I4", "I5", "I6", "I7", "I8"};
        for (int i = 0; i < m; i++) { nome[i] = rot[i]; busca[i] = i + 1; decod[i] = i + 2; }
        diagrama((int)m, nome, busca, decod);
        printf("\n");
    }
    totais(m, K + (m - 1), 0);   /* T = [K + (M - 1)] * t */
    return 0;
}

Observe: leia o diagrama por coluna. No ciclo 5, os cinco estágios estão ocupados, cada um com uma instrução diferente. O total segue a fórmula do livro: T = [K + (M − 1)] × t, em que K é o número de estágios, M o de instruções e t a duração do ciclo. Sem pipeline, cada instrução passaria pelos K estágios sozinha: K × M × t.

Agora o exemplo do livro. Num processador real, as instruções não levam o mesmo tempo. O livro de Silva e Borges supõe que, sem pipeline, cada instrução leva em média 6,5 ns (uma soma, só 6 ns), porque cada uma usa só as etapas de que precisa. Com pipeline, toda instrução passa pelos 5 estágios de 2 ns, mesmo nos que não usa.

Atividade 2 · até 30 min Simulação + medição real

Quando uma conta espera pela outra

Objetivo: ver o que acontece quando uma instrução precisa do resultado da anterior, primeiro na simulação e depois no processador do LIAC.

Conceitos

Dependência de dados (RAW, read after write)
Uma instrução precisa ler um valor que a anterior ainda está calculando. slide 9
Bolha (stall ou bubble)
Ciclo em que a instrução fica parada esperando; o estágio à frente dela fica sem trabalho útil. slide 7 e slide 9
Adiantamento (forwarding ou bypassing)
Atalho no hardware que entrega o resultado da ULA direto para a instrução seguinte, sem esperar a escrita no registrador. slide 10
Cadeia de somas
Sequência em que cada soma usa o resultado da anterior: a = a + x, depois a = a + x de novo... Duas cadeias (a e b) são independentes entre si.

O que fazer

  1. Veja o exemplo do slide 9 no simulador da Atividade 1. O SUB lê $s0, que o ADD acabou de calcular. O OR não depende de ninguém.

  2. Agora no processador real. O programa cadeias faz sempre as mesmas 100.663.296 somas, divididas em 1, 2, 4, 8 ou 12 cadeias independentes. Ele também mede uma régua: uma soma de inteiros em cadeia, que neste tipo de processador leva 1 ciclo. Assim ele converte nanossegundos em ciclos.
    Criar cadeias.c
    Cole no terminal para criar cadeias.c
    cat > cadeias.c <<'FIM'
/* cadeias.c - MEDICAO REAL: o somador do processador tambem e um pipeline.
   Faz sempre o mesmo numero de somas de ponto flutuante (double), mas
   divididas em 1, 2, 4, 8 ou 12 "cadeias". Numa cadeia, cada soma usa o
   resultado da anterior (dependencia de dados). Cadeias diferentes sao
   independentes e podem ocupar o somador ao mesmo tempo.
   Uso: ./cadeias */
#define _POSIX_C_SOURCE 199309L
#include <stdio.h>
#include <time.h>

#define SOMAS (3L << 25)   /* 100.663.296 somas em cada linha (divisivel por 12) */
#define RODADAS 8          /* usa a rodada mais rapida de cada linha */
#define REGUA (1L << 24)   /* somas da regua em cada rodada */

/* Tempo de CPU deste programa: nao conta o tempo em que ele esperou a vez
   enquanto outros programas (de colegas) usavam o processador. */
static double agora(void) {
    struct timespec t;
    clock_gettime(CLOCK_PROCESS_CPUTIME_ID, &t);
    return (double)t.tv_sec + (double)t.tv_nsec * 1e-9;
}

/* Regua: soma de inteiros em cadeia. Cada soma depende da anterior e leva
   1 ciclo de relogio; entao "ns por soma" = duracao de 1 ciclo. Sao 8 somas
   por volta para que o proprio laco (o salto de volta) nao atrase a regua.
   O valor somado vem da memoria (passo), para o processador nao adiantar
   a conta como faria com uma constante. */
static volatile long passo = 1;
#define SOMA1 x = x + y; __asm__ volatile("" : "+r"(x));  /* asm: o compilador */
static double regua_ns(void) {                            /* nao pode pular  */
    long x = 0, y = passo;                                /* nenhuma soma    */
    double t0 = agora();
    for (long i = 0; i < REGUA; i += 8) {
        SOMA1 SOMA1 SOMA1 SOMA1 SOMA1 SOMA1 SOMA1 SOMA1
    }
    return (agora() - t0) / (double)REGUA * 1e9;
}

static volatile double um = 1.0;   /* lido da memoria: o compilador nao adivinha */

/* Cada funcao faz as mesmas SOMAS somas, em 1, 2, 4, 8 ou 12 cadeias.
   Cada cadeia comeca num valor diferente; se todas comecassem iguais, o
   compilador perceberia e faria uma so. O resultado desconta esse inicio.
   Sem "static", o compilador mantem cada funcao separada e simples. */
double cadeia1(double x) {
    double a = 0;
    for (long i = 0; i < SOMAS; i++) { a += x; }
    return a;
}
double cadeia2(double x) {
    double a = 0, b = 1;
    for (long i = 0; i < SOMAS; i += 2) { a += x; b += x; }
    return a + b - 1;
}
double cadeia4(double x) {
    double a = 0, b = 1, c = 2, d = 3;
    for (long i = 0; i < SOMAS; i += 4) { a += x; b += x; c += x; d += x; }
    return a + b + c + d - 6;
}
double cadeia8(double x) {
    double a = 0, b = 1, c = 2, d = 3, e = 4, f = 5, g = 6, h = 7;
    for (long i = 0; i < SOMAS; i += 8) {
        a += x; b += x; c += x; d += x; e += x; f += x; g += x; h += x;
    }
    return a + b + c + d + e + f + g + h - 28;
}
double cadeia12(double x) {
    double a = 0, b = 1, c = 2, d = 3, e = 4, f = 5, g = 6, h = 7;
    double p = 8, q = 9, r = 10, s = 11;
    for (long i = 0; i < SOMAS; i += 12) {
        a += x; b += x; c += x; d += x; e += x; f += x; g += x; h += x;
        p += x; q += x; r += x; s += x;
    }
    return a + b + c + d + e + f + g + h + p + q + r + s - 66;
}

static double somar(int n) {
    if (n == 1) return cadeia1(um);
    if (n == 2) return cadeia2(um);
    if (n == 4) return cadeia4(um);
    if (n == 8) return cadeia8(um);
    return cadeia12(um);
}

int main(void) {
    /* Rodadas intercaladas: cada rodada mede a regua e todas as linhas.
       Se outro programa dividir o mesmo nucleo por um momento, so algumas
       rodadas ficam lentas; usamos a mais rapida de cada medida. */
    const int n[] = {1, 2, 4, 8, 12};
    double tempo[5] = {1e9, 1e9, 1e9, 1e9, 1e9}, resultado[5] = {0};
    double ciclo = 1e9;
    regua_ns();   /* aquecimento: da tempo ao processador de acelerar */
    for (int r = 0; r < RODADAS; r++) {
        double c = regua_ns();
        if (c < ciclo) ciclo = c;
        for (int k = 0; k < 5; k++) {
            double t0 = agora();
            resultado[k] = somar(n[k]);
            double t = agora() - t0;
            if (t < tempo[k]) tempo[k] = t;
        }
    }

    printf("Mesmas %ld somas em cada linha; melhor de %d rodadas.\n", SOMAS, RODADAS);
    printf("Regua: 1 ciclo de relogio = %.3f ns (clock estimado: %.1f GHz)\n\n",
           ciclo, 1.0 / ciclo);
    printf("cadeias   tempo(ms)   ns/soma   ciclos/soma    resultado\n");
    for (int k = 0; k < 5; k++) {
        double ns = tempo[k] / (double)SOMAS * 1e9;
        printf("%7d %11.1f %9.3f %13.2f %12.0f\n", n[k], tempo[k] * 1e3, ns, ns / ciclo,
               resultado[k]);
    }
    return 0;
}

Observe: a coluna resultado é igual em todas as linhas, então o trabalho foi o mesmo. Só mudou quanto as somas dependiam umas das outras. Com 1 cadeia, cada soma espera a anterior terminar: os ciclos por soma mostram a latência do somador. Com várias cadeias, o somador recebe uma soma nova antes de a anterior acabar, como o lava-jato que recebe um carro novo a cada etapa. Repare que o tempo cai pela metade a cada vez que as cadeias dobram, até parar.

Atenção: o processador do LIAC não é o pipeline didático de 5 estágios. Ele tem bem mais estágios, executa mais de uma instrução por ciclo e até muda a ordem delas (capítulo 8 do livro, seções 8.2 e 8.3). O princípio, porém, é o mesmo: uma conta que depende de outra precisa esperar.

Atividade 3 · até 30 min Medição real

O processador tenta adivinhar

Objetivo: medir quanto custa ao pipeline errar a previsão de um desvio.

Conceitos

Desvio condicional (branch)
Instrução que decide se o programa continua na linha seguinte ou pula para outro trecho. Todo if vira um desvio. slide 7
Predição de desvio (branch prediction)
Em vez de parar e esperar a decisão, o processador adivinha o caminho e já busca as próximas instruções. Para adivinhar, usa o histórico dos últimos desvios. slide 10
Esvaziamento (flush) e penalidade
Se a previsão erra, as instruções buscadas pelo caminho errado são descartadas e o pipeline recomeça. Os ciclos perdidos são a penalidade. slide 8
CPI (ciclos por instrução)
Média de ciclos gastos por instrução. Ideal = 1; bolhas e erros de previsão aumentam esse número. slide 12

O que fazer

  1. O programa desvios percorre os mesmos 65.536 números (de 0 a 255) e soma só os que são maiores ou iguais a 128. Esse if é o desvio. Ele faz isso com os números em três ordens: ordenada (crescente), alternada (pequeno, grande, pequeno, grande...) e aleatória.
  2. Crie, compile e rode:
    Criar desvios.c
    Cole no terminal para criar desvios.c
    cat > desvios.c <<'FIM'
/* desvios.c - MEDICAO REAL: o processador tenta adivinhar os desvios.
   Percorre os mesmos 65.536 numeros (0 a 255) em tres ordens e soma
   so os que sao >= 128. O "if" vira uma instrucao de desvio condicional
   (branch). Muda so a ordem: a quantidade de trabalho e a mesma.
   Uso: ./desvios */
#define _POSIX_C_SOURCE 199309L
#include <stdio.h>
#include <time.h>

#define N 65536
#define VOLTAS 200         /* 200 voltas: 13.107.200 desvios por ordem */
#define RODADAS 10         /* usa a rodada mais rapida de cada ordem */
#define REGUA (1L << 24)   /* somas da regua em cada rodada */

static unsigned char v[3][N];   /* os mesmos numeros em tres ordens */
static unsigned semente = 2026;

/* Tempo de CPU deste programa: nao conta o tempo em que ele esperou a vez
   enquanto outros programas (de colegas) usavam o processador. */
static double agora(void) {
    struct timespec t;
    clock_gettime(CLOCK_PROCESS_CPUTIME_ID, &t);
    return (double)t.tv_sec + (double)t.tv_nsec * 1e-9;
}

static unsigned sorteio(void) {   /* gerador simples, sempre a mesma sequencia */
    semente = semente * 1103515245u + 12345u;
    return semente >> 16;
}

/* Regua: soma de inteiros em cadeia. Cada soma depende da anterior e leva
   1 ciclo de relogio; entao "ns por soma" = duracao de 1 ciclo. Sao 8 somas
   por volta para que o proprio laco (o salto de volta) nao atrase a regua.
   O valor somado vem da memoria (passo), para o processador nao adiantar
   a conta como faria com uma constante. */
static volatile long passo = 1;
#define SOMA1 x = x + y; __asm__ volatile("" : "+r"(x));  /* asm: o compilador */
static double regua_ns(void) {                            /* nao pode pular  */
    long x = 0, y = passo;                                /* nenhuma soma    */
    double t0 = agora();
    for (long i = 0; i < REGUA; i += 8) {
        SOMA1 SOMA1 SOMA1 SOMA1 SOMA1 SOMA1 SOMA1 SOMA1
    }
    return (agora() - t0) / (double)REGUA * 1e9;
}

static long percorrer(const unsigned char *a) {
    long soma = 0;
    for (int r = 0; r < VOLTAS; r++)
        for (int i = 0; i < N; i++)
            if (a[i] >= 128) {                 /* o desvio que sera adivinhado */
                soma += a[i];
                __asm__ volatile("");           /* mantem o desvio de verdade */
            }
    return soma;
}

int main(void) {
    /* alternada: pequeno (0-127), grande (128-255), pequeno, grande... */
    int conta[256] = {0}, p = 0;
    for (int i = 0; i < N; i++) {
        v[1][i] = (unsigned char)(sorteio() % 128 + (i % 2 ? 128 : 0));
        v[2][i] = v[1][i];
        conta[v[1][i]]++;
    }
    for (int x = 0; x < 256; x++)                  /* ordenada: crescente */
        while (conta[x]--) v[0][p++] = (unsigned char)x;
    for (int i = N - 1; i > 0; i--) {              /* aleatoria: embaralha */
        int j = (int)(sorteio() % (unsigned)(i + 1));
        unsigned char t = v[2][i]; v[2][i] = v[2][j]; v[2][j] = t;
    }

    /* Rodadas intercaladas: cada rodada mede a regua e as tres ordens.
       Se outro programa dividir o mesmo nucleo por um momento, so algumas
       rodadas ficam lentas; usamos a mais rapida de cada medida. */
    const char *nome[] = {"ordenada", "alternada", "aleatoria"};
    double tempo[3] = {1e9, 1e9, 1e9}, ciclo = 1e9;
    long resultado[3] = {0, 0, 0};
    regua_ns();   /* aquecimento: da tempo ao processador de acelerar */
    for (int r = 0; r < RODADAS; r++) {
        double c = regua_ns();
        if (c < ciclo) ciclo = c;
        for (int k = 0; k < 3; k++) {
            double t0 = agora();
            resultado[k] = percorrer(v[k]);
            double t = agora() - t0;
            if (t < tempo[k]) tempo[k] = t;
        }
    }

    printf("Mesmos %d numeros, %d voltas: %ld desvios por ordem; melhor de %d rodadas.\n",
           N, VOLTAS, (long)N * VOLTAS, RODADAS);
    printf("Regua: 1 ciclo de relogio = %.3f ns (clock estimado: %.1f GHz)\n\n",
           ciclo, 1.0 / ciclo);
    printf("ordem        tempo(ms)   ns/numero   ciclos/numero    resultado\n");
    for (int k = 0; k < 3; k++) {
        double ns = tempo[k] / ((double)N * VOLTAS) * 1e9;
        printf("%-10s %11.1f %11.3f %15.2f %12ld\n", nome[k], tempo[k] * 1e3, ns, ns / ciclo,
               resultado[k]);
    }
    return 0;
}
